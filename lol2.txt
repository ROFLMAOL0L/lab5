#!/bin/bash

set -euo pipefail

IP="${1:?Missing IP}"
FAILURES="${2:-3}"
HOSTNAME="$(hostname)"

source /etc/hadoop-alerts/mail.conf

{
    printf 'From: %s\n' "$SENDER"
    printf 'To: %s\n' "$RECIPIENT"
    printf 'Subject: [Hadoop SECURITY] SSH alert on %s\n' "$HOSTNAME"
    printf 'MIME-Version: 1.0\n'
    printf 'Content-Type: text/plain; charset=UTF-8\n'
    printf '\n'

    printf 'Suspicious SSH activity detected!\n\n'
    printf 'Hostname: %s\n' "$HOSTNAME"
    printf 'Source IP: %s\n' "$IP"
    printf 'Failed attempts: %s\n' "$FAILURES"
    printf 'Threshold: 3 attempts in 5 minutes\n'
    printf 'Time: %s\n' "$(date '+%Y-%m-%d %H:%M:%S %Z')"
    printf '\n'
    printf 'This notification was generated by Fail2ban.\n'
} | /usr/bin/msmtp -t
